{{/* Persistent cassette-future music player. Enumerates static/music/*.mp3 at build, skips anything containing "elevenlabs", shuffles, and resumes across page loads. */}} {{- $titles := site.Data.tracks -}} {{- $tracks := slice -}} {{- range (readDir "static/music") -}} {{- $n := .Name -}} {{- if and (strings.HasSuffix (lower $n) ".mp3") (not (in (lower $n) "elevenlabs")) -}} {{- $stem := strings.TrimSuffix ".mp3" $n -}} {{- $title := index $titles $stem -}} {{- $tracks = $tracks | append (dict "src" (printf "music/%s" $n | relURL) "title" $title) -}} {{- end -}} {{- end -}}
